backend-service/
├── src/
│   ├── api/                  # Các endpoint API (Routes & Controllers)
│   │   ├── controllers/      # Xử lý request/response (CustomerController, BookingController...)
│   │   ├── middlewares/      # Xác thực token (Auth), kiểm tra quyền, validate dữ liệu
│   │   └── routes/           # Định tuyến API (ví dụ: /api/v1/bookings)
│   │
│   ├── core/                 # Chứa Business Logic (Quy tắc nghiệp vụ cốt lõi)
│   │   ├── services/         # Logic ghép cuốc, tính giá tiền, tạo đơn hàng...
│   │   └── sockets/          # Xử lý realtime (Socket.io/Websocket) để theo dõi vị trí/trạng thái
│   │
│   ├── infrastructure/       # Giao tiếp với bên ngoài (Database, 3rd party)
│   │   ├── database/         # Models/Entities (User, Maid, Booking, Payment) & Migrations
│   │   ├── redis/            # Caching & Queue (phục vụ việc matching nhanh chóng)
│   │   └── third_party/      # Tích hợp cổng thanh toán (Momo, VNPay), gửi SMS/Email, Map API
│   │
│   ├── utils/                # Các hàm tiện ích dùng chung (format ngày tháng, tạo mã ngẫu nhiên)
│   └── config/               # Cấu hình môi trường (Database, JWT, API Keys)
│
├── tests/                    # Unit tests & Integration tests
├── .env                      # Biến môi trường
├── Dockerfile                # File đóng gói ứng dụng
└── package.json              # Quản lý thư viện
